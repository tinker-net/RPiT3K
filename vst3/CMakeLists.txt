cmake_minimum_required(VERSION 3.22)
project(TON3K VERSION 0.3.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.8 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(juce)
set(NAM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../NeuralAmpModelerCore" CACHE PATH "Pinned NAM Core checkout")
file(GLOB NAM_SOURCES "${NAM_ROOT}/NAM/*.cpp" "${NAM_ROOT}/NAM/*.c" "${NAM_ROOT}/NAM/*/*.cpp")
if(NOT EXISTS "${NAM_ROOT}/NAM/get_dsp.h")
  message(FATAL_ERROR "Clone pinned NAM Core first; see README")
endif()
set(D "${CMAKE_CURRENT_SOURCE_DIR}/../rpi/vendor/daisysp")
set(TON3K_FORMATS VST3)
if(APPLE)
  list(APPEND TON3K_FORMATS AU)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  list(APPEND TON3K_FORMATS LV2)
endif()
set(TON3K_WEBVIEW2 FALSE)
if(WIN32)
  set(TON3K_WEBVIEW2 TRUE)
endif()
juce_add_plugin(TON3K COMPANY_NAME "TinkerNet" BUNDLE_ID net.tinker.ton3k PLUGIN_MANUFACTURER_CODE Tnkr PLUGIN_CODE T3k1 FORMATS ${TON3K_FORMATS} LV2URI "urn:tinkernet:ton3k" PRODUCT_NAME "TON3K" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE NEEDS_WEB_BROWSER TRUE NEEDS_WEBVIEW2 ${TON3K_WEBVIEW2})
target_sources(TON3K PRIVATE Plugin.cpp ${NAM_SOURCES} "${D}/mit/Effects/chorus.cpp" "${D}/lgpl/Source/Dynamics/compressor.cpp" "${D}/lgpl/Source/Effects/reverbsc.cpp")
target_include_directories(TON3K PRIVATE "${NAM_ROOT}" "${NAM_ROOT}/Dependencies/eigen" "${NAM_ROOT}/Dependencies/nlohmann" "${D}/mit" "${D}/mit/Utility" "${D}/lgpl/Source/Dynamics" "${D}/lgpl/Source/Effects")
target_compile_definitions(TON3K PRIVATE NAM_SAMPLE_FLOAT NAM_ENABLE_A2_FAST JUCE_WEB_BROWSER=1 JUCE_USE_WIN_WEBVIEW2_WITH_STATIC_LINKING=$<BOOL:${WIN32}> JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(TON3K PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(SHARED_EDITOR "${CMAKE_CURRENT_SOURCE_DIR}/../rpi/web.html")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SHARED_EDITOR}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_editor.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_editor.py" "${SHARED_EDITOR}" "${CMAKE_CURRENT_BINARY_DIR}/plugin.html" COMMAND_ERROR_IS_FATAL ANY)
set(SCORE_ASSETS "${CMAKE_CURRENT_BINARY_DIR}/score-assets")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_score_assets.py" "${SCORE_ASSETS}" COMMAND_ERROR_IS_FATAL ANY)
file(GLOB SCORE_FILES "${SCORE_ASSETS}/*")
juce_add_binary_data(Ton3kWeb SOURCES "${CMAKE_CURRENT_BINARY_DIR}/plugin.html" ${SCORE_FILES})
target_link_libraries(TON3K PRIVATE Ton3kWeb juce::juce_gui_extra)
